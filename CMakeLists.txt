cmake_minimum_required(VERSION 3.16)
project(basement_ventilation VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS core/*.cpp)
add_library(ventilation_core ${CORE_SOURCES})
target_include_directories(ventilation_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  add_compile_options(/EHsc /W4 /WX /utf-8)
  target_compile_options(ventilation_core PRIVATE /W4 /WX)
else()
  target_compile_options(ventilation_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_library(ventilation_simulation simulator/BasementModel.cpp simulator/DehumidifierModel.cpp simulator/SimulationConfiguration.cpp simulator/SimulationRunner.cpp simulator/Dashboard.cpp simulator/Calibration.cpp simulator/BatchSimulation.cpp simulator/SweepReport.cpp)
target_link_libraries(ventilation_simulation PUBLIC ventilation_core)
add_executable(pinceszellozteto simulator/main.cpp)
target_link_libraries(pinceszellozteto PRIVATE ventilation_simulation)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS test/test_*.cpp)
foreach(source ${TEST_SOURCES})
  get_filename_component(name ${source} NAME_WE)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE ventilation_core ventilation_simulation)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
add_test(NAME simulator_cli_help COMMAND pinceszellozteto --help)
add_test(NAME simulator_invalid_step COMMAND pinceszellozteto --step 0)
set_tests_properties(simulator_invalid_step PROPERTIES WILL_FAIL TRUE)
add_test(NAME simulator_export COMMAND pinceszellozteto --scenario winter --hours 0.1 --out ${CMAKE_CURRENT_BINARY_DIR}/ctest-export)
add_library(ventilation_display display/DisplaySession.cpp display/DisplayStatus.cpp display/OledUi.cpp)
target_link_libraries(ventilation_display PUBLIC ventilation_simulation)
add_executable(test_display_session test/display_session_test.cpp)
target_link_libraries(test_display_session PRIVATE ventilation_display)
add_test(NAME test_display_session COMMAND test_display_session)
add_executable(test_tank_interlock test/tank_interlock_test.cpp)
target_link_libraries(test_tank_interlock PRIVATE ventilation_display)
add_test(NAME test_tank_interlock COMMAND test_tank_interlock)
add_executable(test_oled_ui test/oled_ui_test.cpp)
target_link_libraries(test_oled_ui PRIVATE ventilation_display)
add_test(NAME test_oled_ui COMMAND test_oled_ui)
if(WIN32)
  add_executable(pince-kijelzo WIN32 display/WindowsDisplay.cpp)
  target_link_libraries(pince-kijelzo PRIVATE ventilation_display gdiplus shell32)
  target_compile_definitions(pince-kijelzo PRIVATE UNICODE _UNICODE NOMINMAX)
  add_test(NAME display_touch_navigation COMMAND pince-kijelzo --ui-test)
  add_test(NAME display_render COMMAND pince-kijelzo --render ${CMAKE_CURRENT_BINARY_DIR}/display-preview.png 5 1)
endif()
