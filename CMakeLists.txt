cmake_minimum_required(VERSION 3.16)
project(basement_ventilation VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS core/*.cpp)
add_library(ventilation_core ${CORE_SOURCES})
target_include_directories(ventilation_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  add_compile_options(/EHsc /W4 /WX /utf-8)
  target_compile_options(ventilation_core PRIVATE /W4 /WX)
else()
  target_compile_options(ventilation_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_library(ventilation_simulation simulator/BasementModel.cpp simulator/DehumidifierModel.cpp simulator/SimulationRunner.cpp simulator/Dashboard.cpp)
target_link_libraries(ventilation_simulation PUBLIC ventilation_core)
add_executable(pinceszellozteto simulator/main.cpp)
target_link_libraries(pinceszellozteto PRIVATE ventilation_simulation)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS test/test_*.cpp)
foreach(source ${TEST_SOURCES})
  get_filename_component(name ${source} NAME_WE)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE ventilation_core ventilation_simulation)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
add_test(NAME simulator_cli_help COMMAND pinceszellozteto --help)
add_test(NAME simulator_invalid_step COMMAND pinceszellozteto --step 0)
set_tests_properties(simulator_invalid_step PROPERTIES WILL_FAIL TRUE)
add_test(NAME simulator_export COMMAND pinceszellozteto --scenario winter --hours 0.1 --out ${CMAKE_CURRENT_BINARY_DIR}/ctest-export)
